#Navigation
#Commands  			#DESCRIPTION
ls                              list files and directories
ls -l				long list file format permission,owner,size,modified_date
ls -a				show hidden files too(starts with dot)
ls lh				long list with file human readable file sizes mb kb gb
cd directory name 		change directory
cd ..				moves up one level to the parent directory
cd ~ or cd			return too the current user home directory
cd - 				switch back to the previous or last directory	
tree				display directory structure as tree
